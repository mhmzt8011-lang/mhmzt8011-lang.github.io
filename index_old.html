🎲 Dice Roller Game

import random
from kivy.app import App
from kivy.uix.boxlayout import BoxLayout
from kivy.uix.label import Label
from kivy.uix.button import Button
from kivy.core.text import LabelBase
from kivy.core.window import Window
import arabic_reshaper
from bidi.algorithm import get_display


# Fix Arabic text rendering (RTL shaping)
def ar(text):
    reshaped_text = arabic_reshaper.reshape(text)
    return get_display(reshaped_text)


# Dice faces
DICE_FACES = {1: '⚀', 2: '⚁', 3: '⚂', 4: '⚃', 5: '⚄', 6: '⚅'}

# Dark blue app background
Window.clearcolor = (0.1, 0.2, 0.3, 1)


class DiceGame(BoxLayout):
    def __init__(self, **kwargs):
        super().__init__(orientation='vertical', padding=30, spacing=20, **kwargs)

        # 1. Game title
        self.title = Label(
            text=ar("لعبة حجر النرد"),
            font_name='ArabicFont',
            font_size=50,
            size_hint=(1, 0.2)
        )
        self.add_widget(self.title)

        # 2. Dice frame (side by side)
        self.dice_layout = BoxLayout(orientation='horizontal', spacing=20, size_hint=(1, 0.4))

        self.dice1 = Label(text='⚀', font_size=150)
        self.dice2 = Label(text='⚀', font_size=150)

        self.dice_layout.add_widget(self.dice1)
        self.dice_layout.add_widget(self.dice2)
        self.add_widget(self.dice_layout)

        # 3. Result label
        self.result_label = Label(
            text=ar("اضغط على الزر للبدء"),
            font_name='ArabicFont',
            font_size=40,
            size_hint=(1, 0.2),
            color=(1, 0.8, 0, 1)
        )
        self.add_widget(self.result_label)

        # 4. Roll button
        self.roll_btn = Button(
            text=ar("ارمي النرد!"),
            font_name='ArabicFont',
            font_size=45,
            size_hint=(1, 0.2),
            background_color=(0, 0.7, 0.3, 1)
        )
        self.roll_btn.bind(on_press=self.roll_dice)
        self.add_widget(self.roll_btn)

    def roll_dice(self, instance):
        val1 = random.randint(1, 6)
        val2 = random.randint(1, 6)
        total = val1 + val2

        self.dice1.text = DICE_FACES[val1]
        self.dice2.text = DICE_FACES[val2]

        if val1 == val2:
            msg = f"مزدوج! المجموع: {total}"
        elif total >= 10:
            msg = f"رمية قوية! المجموع: {total}"
        elif total <= 4:
            msg = f"رمية ضعيفة.. المجموع: {total}"
        else:
            msg = f"النتيجة: {total}"

        self.result_label.text = ar(msg)


class MyApp(App):
    def build(self):
        # NOTE: place a font file named arial.ttf in the same folder as this script
        LabelBase.register(name='ArabicFont', fn_regular='arial.ttf')
        return DiceGame()


if __name__ == '__main__':
    MyApp().run()
